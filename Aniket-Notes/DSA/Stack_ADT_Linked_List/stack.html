📚 What is Stack ADT?

A Stack is a linear data structure following the LIFO (Last In, First Out) principle. The last element pushed onto the stack is always the first element popped off.

In a Linked List implementation, each stack element is stored inside a dynamically allocated Node containing:

  • data: The value stored in the node.
  • next: Pointer to the node underneath it.
  • top: Pointer to the top node (or NULL if empty).

💻 C++ Class Implementation 

#include <iostream>
using namespace std;

class Stack {
private:
    struct Node {
        int data;
        Node* next;
    };
    Node* top;

public:
    Stack() {
        top = nullptr;
    }

    void push(int value) {
        Node* newNode = new Node();
        newNode->data = value;
        newNode->next = top;
        top = newNode;
        cout << "Pushed: " << value << endl;
    }

    void pop() {
        if (top == nullptr) {
            cout << "Stack Underflow!" << endl;
            return;
        }
        Node* temp = top;
        top = top->next;
        cout << "Popped: " << temp->data << endl;
        delete temp;
    }

    void peek() {
        if (top == nullptr) {
            cout << "Stack is Empty!" << endl;
            return;
        }
        cout << "Top Element: " << top->data << endl;
    }

    bool isEmpty() {
        return top == nullptr;
    }

    ~Stack() {
        while (top != nullptr) {
            Node* temp = top;
            top = top->next;
            delete temp;
        }
    }
};

int main() {
    Stack s;
    s.push(10);
    s.push(20);
    s.push(30);

    s.peek();
    s.pop();
    s.peek();

    return 0;
}

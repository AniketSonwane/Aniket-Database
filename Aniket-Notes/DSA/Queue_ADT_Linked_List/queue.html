📚 What is Queue ADT?

A Queue is a linear data structure following the FIFO (First In, First Out) principle. The first element inserted at the rear is always the first element removed from the front.

In a Linked List implementation, each node is dynamically allocated containing:

  • front: Pointer to the first node in the queue.
  • rear: Pointer to the last node in the queue.
  • next: Pointer linking each node to the next node in line (last node points to NULL).

⚙️ Special Pointer Cases

  • 1. First Enqueue (Empty Queue): When enqueueing the very first node, both front and rear are updated to point to the new node (front = rear = newNode).
  • 2. Last Dequeue (Single Element Queue): When dequeueing the last remaining node, after moving front = front->next (NULL), rear must also be explicitly reset to NULL (if (front == NULL) rear = NULL).

💻 C++ Class Implementation 

#include <iostream>
using namespace std;

class Queue {
    struct Node {
        int data;
        Node* next;

        Node(int value) {
            data = value;
            next = NULL;
        }
    };

    Node* front;
    Node* rear;

public:
    Queue() {
        front = NULL;
        rear = NULL;
    }

    // Insert element
    void enqueue(int value) {
        Node* newNode = new Node(value);

        if (rear == NULL) {
            front = rear = newNode;
            return;
        }

        rear->next = newNode;
        rear = newNode;
    }

    // Remove element
    void dequeue() {
        if (front == NULL) {
            cout << "Queue Underflow" << endl;
            return;
        }

        Node* temp = front;
        cout << "Deleted: " << temp->data << endl;

        front = front->next;

        if (front == NULL)
            rear = NULL;

        delete temp;
    }

    // Show front element
    void peek() {
        if (front == NULL) {
            cout << "Queue is Empty" << endl;
            return;
        }

        cout << "Front: " << front->data << endl;
    }

    // Display queue
    void display() {
        if (front == NULL) {
            cout << "Queue is Empty" << endl;
            return;
        }

        Node* temp = front;

        while (temp != NULL) {
            cout << temp->data << " ";
            temp = temp->next;
        }

        cout << endl;
    }
};

int main() {
    Queue q;

    q.enqueue(10);
    q.enqueue(20);
    q.enqueue(30);
    q.enqueue(40);

    cout << "Queue: ";
    q.display();

    q.peek();

    q.dequeue();
    q.dequeue();

    cout << "Queue after dequeue: ";
    q.display();

    return 0;
}
